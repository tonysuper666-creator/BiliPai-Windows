function(verify_manifest manifest expected root count)
  file(SHA256 "${manifest}" manifest_sha)
  if(NOT manifest_sha STREQUAL expected)
    message(FATAL_ERROR "Pinned manifest changed: ${manifest}")
  endif()
  file(READ "${manifest}" pins)
  string(JSON actual_count LENGTH "${pins}")
  if(NOT actual_count EQUAL count)
    message(FATAL_ERROR "Incomplete pinned closure")
  endif()
  math(EXPR last "${count}-1")
  foreach(i RANGE 0 ${last})
    string(JSON path GET "${pins}" ${i} path)
    string(JSON expected_sha GET "${pins}" ${i} sha256)
    if(IS_ABSOLUTE "${path}" OR path MATCHES "(^|/)\\.\\.(/|$)" OR path MATCHES "[:\\\\]")
      message(FATAL_ERROR "Invalid pinned source path")
    endif()
    file(SHA256 "${root}/${path}" actual_sha)
    string(TOLOWER "${expected_sha}" expected_sha)
    if(NOT actual_sha STREQUAL expected_sha)
      message(FATAL_ERROR "Fixed source bytes changed: ${path}")
    endif()
  endforeach()
endfunction()

foreach(i RANGE 0 19)
  string(JSON path GET "${union}" ${i} path)
  string(JSON closure GET "${union}" ${i} sourceClosure)
  string(JSON expected_sha GET "${union}" ${i} sha256)
  if(IS_ABSOLUTE "${path}" OR path MATCHES "(^|/)\\.\\.(/|$)" OR path MATCHES "[:\\\\]")
    message(FATAL_ERROR "Invalid union path")
  endif()
  if(closure STREQUAL "veyra-core")
    set(root "${VEYRA_FIXED_SOURCE}")
  elseif(closure STREQUAL "veyra-dlss-sr")
    set(root "${VEYRA_DLSS_FIXED_SOURCE}")
  else()
    message(FATAL_ERROR "Invalid union source closure")
  endif()
  file(SHA256 "${root}/${path}" actual_sha)
  if(NOT actual_sha STREQUAL expected_sha)
    message(FATAL_ERROR "Fixed union hash mismatch")
  endif()
  if(path MATCHES "^include/(veyra/.*\\.h)$")
    set(header "${CMAKE_MATCH_1}")
    configure_file("${root}/${path}" "${fixed_include}/${header}" COPYONLY)
    list(APPEND expected_headers "${fixed_include}/${header}")
  endif()
endforeach()
