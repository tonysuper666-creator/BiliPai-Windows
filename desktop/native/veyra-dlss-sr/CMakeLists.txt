function(verify_source_manifest name expected root count)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${name}" actual_manifest)
  if(NOT actual_manifest STREQUAL expected)
    message(FATAL_ERROR "Pinned manifest itself changed: ${name}")
  endif()
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/${name}" pins)
  string(JSON actual_count LENGTH "${pins}")
  if(NOT actual_count EQUAL count)
    message(FATAL_ERROR "Incomplete source closure: ${name}")
  endif()
  math(EXPR last "${actual_count}-1")
  foreach(i RANGE 0 ${last})
    string(JSON path GET "${pins}" ${i} path)
    string(JSON expected_sha GET "${pins}" ${i} sha256)
    if(IS_ABSOLUTE "${path}" OR path MATCHES "(^|/)\\.\\.(/|$)" OR path MATCHES "[:\\\\]")
      message(FATAL_ERROR "Invalid pinned source path: ${path}")
    endif()
    if(NOT EXISTS "${root}/${path}")
      message(FATAL_ERROR "Missing pinned source: ${path}")
    endif()
    file(SHA256 "${root}/${path}" actual_sha)
    if(NOT actual_sha STREQUAL expected_sha)
      message(FATAL_ERROR "Pinned source/header hash mismatch: ${path}")
    endif()
  endforeach()
endfunction()
